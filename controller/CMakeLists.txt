# Mac:  cmake -S controller -B build && cmake --build build && ctest --test-dir build
# Pi:   같은 명령 (QNX 위 clang). QNX 프로세스 P2~P4는 항상 (P3는 LightGBM 경로를 주면 V1 모델도). P1, P5, P6, rf_stub은 QNX면 항상,
#       P7은 SQLite가 있으면 (third_party/sqlite 또는 시스템)
cmake_minimum_required(VERSION 3.16)
project(mermaid_controller CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra)

add_library(mmcommon STATIC common/features.cpp common/policy.cpp common/store.cpp common/imu.cpp common/sample_ring.cpp)
target_include_directories(mmcommon PUBLIC common)

set(MM_CONFIG_DIR "${CMAKE_CURRENT_SOURCE_DIR}/config")

add_executable(featurize tools/featurize.cpp)
target_link_libraries(featurize mmcommon)

add_executable(policy_replay tools/policy_replay.cpp)
target_link_libraries(policy_replay mmcommon)
target_compile_definitions(policy_replay PRIVATE MM_CONFIG_DIR="${MM_CONFIG_DIR}")

add_executable(calib_apply tools/calib_apply.cpp)
target_link_libraries(calib_apply mmcommon)

# P7 로거 코어 (SQLite). controller/third_party/sqlite/sqlite3.c(공식 amalgamation)가 있으면 함께 빌드하고 (보드),
# 없으면 시스템 SQLite를 쓴다 (Mac SDK). 둘 다 없으면 P7을 빌드하지 않는다
set(MM_SQLITE "")
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/third_party/sqlite/sqlite3.c)
  enable_language(C)
  add_library(mmsqlite STATIC third_party/sqlite/sqlite3.c)
  target_include_directories(mmsqlite PUBLIC third_party/sqlite)
  target_compile_definitions(mmsqlite PRIVATE SQLITE_THREADSAFE=0 SQLITE_OMIT_LOAD_EXTENSION SQLITE_DEFAULT_FOREIGN_KEYS=1)
  set(MM_SQLITE mmsqlite)
else()
  find_package(SQLite3)
  if(TARGET SQLite3::SQLite3)
    set(MM_SQLITE SQLite3::SQLite3)
  elseif(SQLite3_FOUND)
    set(MM_SQLITE SQLite::SQLite3)  # CMake 3.28 전 이름
  endif()
endif()
if(MM_SQLITE)
  add_library(mmlog STATIC common/ride_log.cpp)
  target_include_directories(mmlog PUBLIC common qnx)
  target_link_libraries(mmlog PUBLIC mmcommon ${MM_SQLITE})
  add_executable(test_ride_log tests/test_ride_log.cpp)
  target_link_libraries(test_ride_log mmlog)
  add_test(NAME ride_log COMMAND test_ride_log)
  # 주행 파일 meta.git_commit
  execute_process(COMMAND git describe --always --dirty WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
                  OUTPUT_VARIABLE MM_GIT_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  if(NOT MM_GIT_COMMIT)
    set(MM_GIT_COMMIT "unknown")
  endif()
else()
  message(STATUS "SQLite not found (no third_party/sqlite/sqlite3.c, no system SQLite3): skipping P7 logger")
endif()

# Pi 없이 P5·P6 프로세스를 함께 돌리는 시험 (Mac, Linux): QNX 메시지 전달은 tests/qnxsim이 Unix 소켓으로 흉내,
# UART는 가상 터미널, ESP32는 tests/rf_sim_test.py. 프로세스 코드는 Pi와 같은 qnx/*.cpp. 약 1분 걸림
if(NOT CMAKE_SYSTEM_NAME STREQUAL "QNX")
  find_package(Threads REQUIRED)
  add_library(qnxsim STATIC tests/qnxsim/qnxsim.cpp)
  target_include_directories(qnxsim PUBLIC tests/qnxsim)
  target_link_libraries(qnxsim PUBLIC Threads::Threads)
  foreach(p p5_devmgr p6_comm rf_stub)
    add_executable(${p}_sim qnx/${p}.cpp)
    target_link_libraries(${p}_sim mmrf qnxsim)
  endforeach()
  # P1, P2, P4 (진짜 코드)와 P3 대역(fake_p3, LightGBM 없이)으로 P1→P6 전체를 돌리는 시험
  foreach(p p1_acq p2_feat p4_policy)
    add_executable(${p}_sim qnx/${p}.cpp)
    target_include_directories(${p}_sim PRIVATE qnx)
    target_link_libraries(${p}_sim mmcommon qnxsim)
  endforeach()
  target_link_libraries(p4_policy_sim mmctx)
  # 진짜 P3 (LightGBM 없이 빌드하면 V3 ctx_gru 모델만)
  add_executable(p3_infer_sim qnx/p3_infer.cpp)
  target_include_directories(p3_infer_sim PRIVATE qnx)
  target_link_libraries(p3_infer_sim mmctx qnxsim)
  add_executable(fake_p3_sim tests/fake_p3.cpp)
  target_include_directories(fake_p3_sim PRIVATE qnx)
  target_link_libraries(fake_p3_sim mmcommon qnxsim)
  if(MM_SQLITE)
    add_executable(p7_logger_sim qnx/p7_logger.cpp)
    target_link_libraries(p7_logger_sim mmlog qnxsim)
    target_compile_definitions(p7_logger_sim PRIVATE MM_GIT_COMMIT="${MM_GIT_COMMIT}")
  endif()
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME rf_sim COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/rf_sim_test.py ${CMAKE_CURRENT_BINARY_DIR})
    set_tests_properties(rf_sim PROPERTIES TIMEOUT 240)
    add_test(NAME pipeline_rf_sim COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/pipeline_rf_sim_test.py ${CMAKE_CURRENT_BINARY_DIR})
    set_tests_properties(pipeline_rf_sim PROPERTIES TIMEOUT 240)
    # P1 LIVE를 가짜 IMU로: 보정 → 주행 → P2 → P3 대역 → P4 (약 30초)
    add_test(NAME live_fake_sim COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/live_fake_sim_test.py ${CMAKE_CURRENT_BINARY_DIR})
    set_tests_properties(live_fake_sim PROPERTIES TIMEOUT 120)
    # V3 문맥 모델로 P1 → P2 → 진짜 P3 → P4 두 주행 (토큰 저장 → 다음 주행 문맥). 모델 폴더를 줄 때만. 약 1분
    set(MM_CTX_MODEL "" CACHE PATH "ctx_gru model folder (models.json kind ctx_gru) for the ctx_sim test")
    if(MM_CTX_MODEL)
      add_test(NAME ctx_sim COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/ctx_sim_test.py ${CMAKE_CURRENT_BINARY_DIR} ${MM_CTX_MODEL})
      set_tests_properties(ctx_sim PROPERTIES TIMEOUT 240)
    endif()
    if(MM_SQLITE)
      # P7 로거: 주행 파일 내용과 격리 (P7 kill -9, SIGSTOP에도 P1~P4는 그대로). 약 1분 30초
      add_test(NAME log_sim COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/log_sim_test.py ${CMAKE_CURRENT_BINARY_DIR})
      set_tests_properties(log_sim PROPERTIES TIMEOUT 240)
      # 폰과 오간 사건(P6·P5 → P7)이 주행 파일 ctx·label 표에 남는지, P7이 멈추거나 죽어도 폰이 보는 것은 같은지. 약 1분 40초
      add_test(NAME log_rf_sim COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/log_rf_sim_test.py ${CMAKE_CURRENT_BINARY_DIR})
      set_tests_properties(log_rf_sim PROPERTIES TIMEOUT 300)
      # 폰 GPS(M09) → gps 표·P2 속도 피처, 위험 확률(M10 RISK) → 폰 (#18, #19). 약 50초
      add_test(NAME gps_risk_sim COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/gps_risk_sim_test.py ${CMAKE_CURRENT_BINARY_DIR})
      set_tests_properties(gps_risk_sim PROPERTIES TIMEOUT 180)
    endif()
  endif()
endif()

# P3 추론 (lib_lightgbm 4.6.0). 경로를 주면 빌드:
#   Pi:  -DMM_LIGHTGBM_INCLUDE=$HOME/mermaid/LightGBM/include -DMM_LIGHTGBM_LIB=$HOME/mermaid/LightGBM/lib_lightgbm.so
#   Mac: pip lightgbm의 lib/lib_lightgbm.dylib + 같은 버전 헤더
set(MM_LIGHTGBM_INCLUDE "" CACHE PATH "LightGBM include directory (contains LightGBM/c_api.h)")
set(MM_LIGHTGBM_LIB "" CACHE FILEPATH "lib_lightgbm shared library")
if(MM_LIGHTGBM_INCLUDE AND MM_LIGHTGBM_LIB)
  add_library(mminfer STATIC common/inference.cpp)
  target_include_directories(mminfer PUBLIC ${MM_LIGHTGBM_INCLUDE})
  target_link_libraries(mminfer PUBLIC mmcommon ${MM_LIGHTGBM_LIB})
  get_filename_component(MM_LIGHTGBM_LIBDIR ${MM_LIGHTGBM_LIB} DIRECTORY)
  add_executable(verify_model tools/verify_model.cpp)
  target_link_libraries(verify_model mminfer)
  set_target_properties(verify_model PROPERTIES BUILD_RPATH ${MM_LIGHTGBM_LIBDIR})
else()
  message(STATUS "MM_LIGHTGBM_INCLUDE / MM_LIGHTGBM_LIB not set: P3 runs only V3 ctx_gru models (no LightGBM)")
endif()
